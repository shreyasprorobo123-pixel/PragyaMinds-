<script>
/* =====================
   USER LOGIN SYSTEM
===================== */
if (!localStorage.getItem("users")) {
  localStorage.setItem("users", JSON.stringify([]));
}

function userRegister() {

  let u = loginUser.value;
  let p = loginPass.value;
  let role = loginRole.value;
  let secret = adminSecret.value;

  // Username + Password required
  if (!u || !p) {
    loginMsg.innerHTML = "❌ Enter username + password!";
    return;
  }

  // ✅ Password length validation
  if (p.length < 6) {
    loginMsg.innerHTML = "❌ Password must be at least 6 characters!";
    return;
  }

  // 🔐 Admin Secret Protection
  if (role === "Admin") {
    if (secret !== "SMART2026") {
      loginMsg.innerHTML = "❌ Wrong Admin Secret Code!";
      return;
    }
  }

  let users = JSON.parse(localStorage.getItem("users"));

  // Prevent duplicate usernames
  if (users.find(x => x.user === u)) {
    loginMsg.innerHTML = "⚠ Username already exists!";
    return;
  }

  // Save user
  users.push({ user: u, pass: p, role: role });
  localStorage.setItem("users", JSON.stringify(users));

  loginMsg.innerHTML =
    "✅ Registered successfully as " + role + "! Now Login.";
}

function userLogin() {
  let u = loginUser.value;
  let p = loginPass.value;
  let role = loginRole.value;

  let users = JSON.parse(localStorage.getItem("users"));
  let found = users.find(x=>x.user===u && x.pass===p && x.role===role);

  if(found){
    localStorage.setItem("loggedInUser", u);
    localStorage.setItem("loggedInRole", role);

    loginScreen.style.display="none";
    mainSite.style.display="block";

    profileCard.style.display="block";
    profileUser.innerHTML=u;
    profileRole.innerHTML=role;

    setTimeout(()=>initMap(),300);

  } else loginMsg.innerHTML="❌ Wrong login!";
}

function userLogout() {
  localStorage.clear();
  location.reload();
}

let futureOn=false;

  futureOn=!futureOn;
  document.body.classList.toggle("futureMode");

  if(futureOn){

/* =====================
   MAP INIT + INCIDENTS
===================== */
let map, heatMap, heatLayer;
let userMarker, routeLine, ambulanceMarker;
let selectedIncidentLocation=null;
let incidentMarkers=[];

function initMap(){
  if(map) return;

  map=L.map("map").setView([20.5937,78.9629],5);
  L.tileLayer("https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png").addTo(map);

  heatMap=L.map("heatmap").setView([20.5937,78.9629],5);
  L.tileLayer("https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png").addTo(heatMap);

  map.on("click",e=>{
    selectedIncidentLocation=e.latlng;
    alert("📍 Incident location selected!");
  });

  loadIncidents();
}

/* INCIDENT SAVE */
function saveIncident(){
  let role=localStorage.getItem("loggedInRole");
  if(role!=="Citizen" && role!=="Admin") return;

  let text=incidentInput.value;
  if(!selectedIncidentLocation) return alert("Click map first!");
  if(!text.trim()) return alert("Enter description!");

  let incidents=JSON.parse(localStorage.getItem("incidents"))||[];
  incidents.push({text,lat:selectedIncidentLocation.lat,lon:selectedIncidentLocation.lng});
  localStorage.setItem("incidents",JSON.stringify(incidents));
  loadIncidents();
}

/* LOAD INCIDENTS + HEATMAP */
function loadIncidents(){
  let incidents=JSON.parse(localStorage.getItem("incidents"))||[];

  incidentList.innerHTML=incidents.map(i=>"🚨 "+i.text+"<hr>").join("");

  incidentMarkers.forEach(m=>map.removeLayer(m));
  incidentMarkers=[];

  let heatPoints=[];

  incidents.forEach(i=>{
    let m=L.marker([i.lat,i.lon]).addTo(map).bindPopup(i.text);
    incidentMarkers.push(m);
    heatPoints.push([i.lat,i.lon,1]);
  });

  if(heatLayer) heatMap.removeLayer(heatLayer);
  heatLayer=L.heatLayer(heatPoints,{radius:35}).addTo(heatMap);

  incidentCount.innerHTML="Total Incidents: "+incidents.length;
}

/* CLEAR INCIDENTS (ADMIN ONLY) */
function clearAllIncidents(){
  let role=localStorage.getItem("loggedInRole");
  if(role!=="Admin") return alert("❌ Only Admin can delete incidents!");

  localStorage.removeItem("incidents");
  loadIncidents();
}

/* SOS */
function triggerSOS(){
  if(!userMarker) return alert("Set location first!");
  sosStatus.innerHTML="🚨 SOS Activated!";

  let pos=userMarker.getLatLng();
  let incidents=JSON.parse(localStorage.getItem("incidents"))||[];
  incidents.push({text:"🆘 SOS Emergency!",lat:pos.lat,lon:pos.lng});
  localStorage.setItem("incidents",JSON.stringify(incidents));
  loadIncidents();

  serviceSelect.value="hospital";
  routeToService();
}

/* LOCATION */
async function setLocationManual(){
  let addr=locationBox.value;
  let data=await (await fetch(`https://nominatim.openstreetmap.org/search?format=json&q=${addr}`)).json();
  if(!data.length) return alert("Not found!");
  setLocation(data[0].lat,data[0].lon,addr);
}

function detectLocation(){
  navigator.geolocation.getCurrentPosition(pos=>{
    setLocation(pos.coords.latitude,pos.coords.longitude,"Auto Detected");
  });
}

function setLocation(lat,lon,name){
  map.setView([lat,lon],13);
  if(userMarker) map.removeLayer(userMarker);
  userMarker=L.marker([lat,lon]).addTo(map).bindPopup(name).openPopup();
  locationData.innerHTML="✅ "+name;
}

/* ROUTING */

async function routeToService() {

  if (!userMarker) {
    alert("📍 Please set your location first!");
    return;
  }

  let type = serviceSelect.value;

  let searchNames = {
    hospital: "hospital",
    police: "police",
    fire_station: "fire station"
  };

  let colors = {
    hospital: "blue",
    police: "red",
    fire_station: "orange"
  };

  let u = userMarker.getLatLng();

  routeInfo.innerHTML = "🔍 Searching nearest " + searchNames[type] + "...";

  try {
    // ✅ Proper Bounding Box Search (5km radius)
    let delta = 0.05; // ~5km

    let url =
      `https://nominatim.openstreetmap.org/search?` +
      `format=json&q=${searchNames[type]}` +
      `&bounded=1` +
      `&viewbox=` +
      `${u.lng - delta},${u.lat + delta},` +
      `${u.lng + delta},${u.lat - delta}` +
      `&limit=1`;

    let results = await (await fetch(url)).json();

    if (!results.length) {
      routeInfo.innerHTML = "❌ No nearby service found!";
      return;
    }

    let dest = results[0];
    let destLat = parseFloat(dest.lat);
    let destLon = parseFloat(dest.lon);

    // Marker for nearest service
    L.marker([destLat, destLon])
      .addTo(map)
      .bindPopup("✅ Nearest " + searchNames[type])
      .openPopup();

    // Straight route demo (reliable)
    let coords = [
      [u.lat, u.lng],
      [destLat, destLon]
    ];

    if (routeLine) map.removeLayer(routeLine);

    routeLine = L.polyline(coords, {
      color: colors[type],
      weight: 6
    }).addTo(map);

    // Distance + ETA
    let distKm =
      (map.distance([u.lat, u.lng], [destLat, destLon]) / 1000).toFixed(2);

    let etaMin = (distKm / 0.8).toFixed(1);

    routeInfo.innerHTML =
      `✅ Found: ${dest.display_name}<br>` +
      `📏 Distance: ${distKm} km<br>` +
      `⏱ ETA: ${etaMin} mins`;

    animateAmbulance(coords);

  }
  catch (err) {
    console.error(err);
    routeInfo.innerHTML =
      "❌ Search failed. Public APIs may be blocked on your network.";
  }
}
